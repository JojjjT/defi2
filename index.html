  const dt = 1 / 252;
  const sum = a => a.reduce((s, x) => s + x, 0);
  const mean = a => sum(a) / a.length;
  const sd = a => { const m = mean(a); return Math.sqrt(sum(a.map(x => (x - m) ** 2)) / (a.length - 1)); };
  const quantile = (s, q) => { const p = (s.length - 1) * q, i = Math.floor(p), j = Math.min(i + 1, s.length - 1); return s[i] + (s[j] - s[i]) * (p - i); };
  const iso = d => new Date(d).toISOString().slice(0, 10);
  const bisect = (a, v, right) => {
    let lo = 0, hi = a.length;
    while (lo < hi) { const m = (lo + hi) >> 1; if (right ? a[m] <= v : a[m] < v) lo = m + 1; else hi = m; }
    return lo;
  };
  const between = (from, to) => {
    const i1 = Math.min(Math.max(bisect(dates, iso(to), true) - 1, 30), dates.length - 1);
    return [Math.max(Math.min(bisect(dates, iso(from), false), i1 - 30), 0), i1];
  };
  const bdays = (start, n) => {
    const out = [start];
    let d = new Date(start + "T00:00:00Z");
    while (out.length < n + 1) { d = new Date(d.getTime() + 864e5); const w = d.getUTCDay(); if (w !== 0 && w !== 6) out.push(iso(d)); }
    return out;
  };
  const rng = seed => {
    let a = seed >>> 0;
    const u = () => { a = (a + 0x6D2B79F5) >>> 0; let t = Math.imul(a ^ (a >>> 15), a | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
    const z = () => Math.sqrt(-2 * Math.log(1 - u())) * Math.cos(2 * Math.PI * u());
    const pois = l => { const L = Math.exp(-l); let k = 0, p = u(); while (p > L) { k++; p *= u(); } return k; };
    return {z, pois};
  };
  const fmt = x => !isFinite(x) ? "—" : (x !== 0 && (Math.abs(x) >= 1e5 || Math.abs(x) < 1e-3)) ? x.toExponential(2) : x.toFixed(4);
  const ranges = [1, 3, 6].map(m => ({count: m, label: `${m}m`, step: "month", stepmode: "backward"}))
    .concat([{count: 1, label: "1a", step: "year", stepmode: "backward"}, {step: "all", label: "todo"}]);

  const plot = (data, layout = {}, height = 440) => {
    const div = document.createElement("div");
    div.style.width = "100%";
    const calcH = () => div.clientWidth < 520 ? Math.max(300, Math.round(height * 0.75)) : height;
    let drawn = false;
    const render = () => {
      if (!div.clientWidth) return;               // pestaña oculta o aún sin ancho: se dibuja al tener tamaño
      if (!drawn) {
        drawn = true;
        Plotly.newPlot(div, data, {height: calcH(), margin: {l: 55, r: 15, t: 20, b: 40}, hovermode: "x unified", legend: {orientation: "h", y: -0.15}, ...layout}, {responsive: true, displaylogo: false});
      } else Plotly.relayout(div, {autosize: true, height: calcH()});
    };
    if (window.ResizeObserver) new ResizeObserver(render).observe(div);
    else { let t = 0; const poll = () => { render(); if (!drawn && t++ < 600) requestAnimationFrame(poll); }; requestAnimationFrame(poll); }
    return div;
  };
  const table = (head, rows) => {
    const d = document.createElement("div");
    d.innerHTML = `<table class="table table-sm"><thead><tr>${head.map(h => `<th>${h}</th>`).join("")}</tr></thead><tbody>${rows.map(r => `<tr>${r.map(c => `<td>${c}</td>`).join("")}</tr>`).join("")}</tbody></table>`;
    return d;
  };

  const calibAR = x => {
    const X = x.slice(0, -1), Y = x.slice(1), n = X.length, mx = mean(X), my = mean(Y);
    let sxy = 0, sxx = 0, sst = 0;
    for (let i = 0; i < n; i++) { sxy += (X[i] - mx) * (Y[i] - my); sxx += (X[i] - mx) ** 2; sst += (Y[i] - my) ** 2; }
    const b = sxy / sxx, a = my - b * mx;
    const sse = sum(Y.map((y, i) => (y - a - b * X[i]) ** 2));
    const reverting = b > 0 && b < 1;
    const kappa = reverting ? -Math.log(b) / dt : 1e-3;
    const level = reverting ? a / (1 - b) : mx;
    const s2 = sse / (n - 2);
    const sigma = reverting ? Math.sqrt(s2 * 2 * kappa / (1 - b * b)) : Math.sqrt(s2 / dt);
    return {X, Y, n, a, b, r2: 1 - sse / sst, reverting, kappa, level, sigma, halfLife: Math.log(2) / kappa * 252, lo: Math.min(...X), hi: Math.max(...X)};
  };
  const calibMerton = (logp, k) => {
    const r = logp.slice(1).map((v, i) => v - logp[i]);
    let m = mean(r), s = sd(r);
    for (let it = 0; it < 20; it++) {
      const d = r.filter(x => Math.abs(x - m) <= k * s);
      m = mean(d); s = sd(d);
    }
    const jump = r.map(x => Math.abs(x - m) > k * s);
    const J = r.filter((x, i) => jump[i]).map(x => x - m);
    const sigma = s / Math.sqrt(dt), lambda = J.length / (r.length * dt);
    const muJ = J.length ? mean(J) : 0, sigJ = J.length > 1 ? sd(J) : 0.02;
    const kk = Math.exp(muJ + sigJ ** 2 / 2) - 1;
    return {r, jump, m, s, k, sigma, lambda, muJ, sigJ, nJ: J.length, mu: m / dt + sigma ** 2 / 2 + lambda * kk};
  };

  const simMR = (x0, kappa, level, sigma, steps, n, seed, tf = x => x) => {
    const R = rng(seed), k = Math.max(kappa, 1e-6), e = Math.exp(-k * dt), s = sigma * Math.sqrt((1 - e * e) / (2 * k));
    return Array.from({length: n}, () => {
      const p = new Float64Array(steps + 1); p[0] = x0;
      for (let i = 0; i < steps; i++) p[i + 1] = level + (p[i] - level) * e + s * R.z();
      return p.map(v => tf(v));
    });
  };
  const simMerton = (x0, q, steps, n, seed) => {
    const R = rng(seed), kk = Math.exp(q.muJ + q.sigJ ** 2 / 2) - 1;
    const drift = (q.mu - q.sigma ** 2 / 2 - q.lambda * kk) * dt, vol = q.sigma * Math.sqrt(dt);
    return Array.from({length: n}, () => {
      const p = new Float64Array(steps + 1); p[0] = x0;
      for (let i = 0; i < steps; i++) {
        const N = R.pois(q.lambda * dt);
        p[i + 1] = p[i] + drift + vol * R.z() + (N ? N * q.muJ + Math.sqrt(N) * q.sigJ * R.z() : 0);
      }
      return p.map(v => Math.exp(v));
    });
  };
  const fan = paths => {
    const out = {mean: [], lo: [], hi: []};
    for (let i = 0; i < paths[0].length; i++) {
      const col = Float64Array.from(paths, p => p[i]).sort();
      out.mean.push(mean(col)); out.lo.push(quantile(col, 0.05)); out.hi.push(quantile(col, 0.95));
    }
    return out;
  };

  const simFigure = ({ticker, i0, i1, paths, level, levelName}) => {
    const steps = paths[0].length - 1, h = prices[ticker], fut = bdays(dates[i1], steps), f = fan(paths), px = [], py = [];
    paths.slice(0, 30).forEach(p => { for (let i = 0; i <= steps; i++) { px.push(fut[i]); py.push(p[i]); } px.push(null); py.push(null); });
    const data = [
      {x: dates.slice(0, i1 + 1), y: h.slice(0, i1 + 1), name: "Real (ventana de calibración)", line: {color: "black", width: 1.5}},
      {x: dates.slice(i1), y: h.slice(i1), name: "Real (posterior)", line: {color: "#d62728", width: 1.5}},
      {x: px, y: py, name: "Trayectorias (30)", line: {color: "rgba(120,120,120,0.35)", width: 0.7}, hoverinfo: "skip"},
      {x: fut, y: f.hi, name: "P95", line: {width: 0}, showlegend: false},
      {x: fut, y: f.lo, name: "Banda P5–P95", fill: "tonexty", fillcolor: "rgba(31,119,180,0.2)", line: {width: 0}},
      {x: fut, y: f.mean, name: "Media simulada", line: {color: "#1f77b4", width: 2}}
    ];
    if (level != null) data.push({x: [dates[i0], fut[steps]], y: [level, level], name: levelName, line: {color: "green", dash: "dash", width: 1}});
    return plot(data, {
      yaxis: {title: "USD"},
      xaxis: {range: [dates[Math.max(i0, i1 - 504)], fut[steps]], rangeselector: {buttons: ranges}},
      shapes: [{type: "rect", xref: "x", yref: "paper", x0: dates[i0], x1: dates[i1], y0: 0, y1: 1, fillcolor: "rgba(0,0,0,0.06)", line: {width: 0}, layer: "below"}],
      annotations: [{x: dates[i0], y: 1, xref: "x", yref: "paper", text: "ventana de calibración", showarrow: false, xanchor: "left", yanchor: "bottom"}]
    }, 500);
  };
  const arFigure = (c, xl, yl) => {
    const span = c.hi - c.lo, ends = [c.lo, c.hi];
    const data = [
      {x: c.X, y: c.Y, mode: "markers", name: "Pares (xₜ, xₜ₊₁)", marker: {size: 4, color: "rgba(80,80,80,0.5)"}},
      {x: ends, y: ends.map(v => c.a + c.b * v), mode: "lines", name: "Ajuste MCO", line: {color: "#d62728", width: 2}},
      {x: ends, y: ends, mode: "lines", name: "xₜ₊₁ = xₜ", line: {color: "black", dash: "dash", width: 1}}

Modelo: dP = κ(θ − P)dt + σ dW sobre el precio. Calibración: regresión MCO Pₜ₊₁ = a + b·Pₜ + ε sobre la ventana elegida; de ahí κ = −ln(b)/Δt, θ = a/(1−b), σ = s_ε·√(2κ/(1−b²)). El punto verde es donde la recta ajustada corta la diagonal: el nivel de equilibrio θ. Simulación: esquema exacto del proceso desde el último precio de la ventana; si hay datos reales posteriores se dibujan en rojo para evaluar el ajuste fuera de muestra. Si b ≥ 1 no hay reversión a la media y κ, θ no son identificables.

Modelo: dS/S = (μ − λk)dt + σ dW + (J − 1)dN, con N Poisson de intensidad λ y ln J ~ N(μ_J, σ_J²). Calibración (detección de saltos): se parte de la media y desviación de los retornos log; se marcan como saltos los retornos a más de k·σ de la media, se re-estiman media y σ sin ellos y se repite hasta converger. Con los saltos detectados: λ = n_saltos / años, μ_J y σ_J = media y desviación del tamaño de los saltos, σ = desviación de la parte difusiva (anualizada) y μ recupera la deriva incluyendo la compensación λk. Subir el umbral detecta menos saltos.

Modelo sobre el logaritmo del precio: d ln P = θ(μ − ln P)dt + σ dW, es decir, el precio nunca es negativo y revierte a exp(μ). Calibración: regresión MCO xₜ₊₁ = a + b·xₜ + ε con x = ln P; θ = −ln(b)/Δt, μ = a/(1−b), σ = s_ε·√(2θ/(1−b²)). Simulación: esquema exacto del proceso desde el último precio de la ventana, transformado con exp. En series con tendencia b ≈ 1 y la reversión es débil o inexistente: la calibración lo muestra tal cual.

  ticker: Inputs.select(tickers, {label: "Ticker"}),

cO = {
  const [i0, i1] = lib.between(sO.from, sO.to);
  return {i0, i1, ...lib.calibAR(prices[sO.ticker].slice(i0, i1 + 1).map(v => Math.log(v)))};
}
